// Generated GAME code/data. This never belongs in an exchangeable Gua DLL.
#nullable enable
using System;
using System.IO;
using System.Linq;
using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;
using System.Runtime.Loader;
using System.Security.Cryptography;
using System.Collections.Generic;

internal static class GuaGameNativePolicy
{
    internal const string RemoteMode = /*MODE*/;
    internal const string ProviderSource = /*SOURCE*/;
    internal const string ProviderBuild = /*BUILD*/;
    private const string StateKey = "Gua.GameOwned.GodotPolicy";
    private sealed record ExpectedFile(string Name,string Sha256);
    private static readonly ExpectedFile[] Files = {
/*EXPECTED_FILES*/
    };
    private static bool verified;
    private static string stage="platform";

    [ModuleInitializer]
    internal static void Initialize()
    {
        try {
            OutputDebugString("GameSource: before first policy call.");
            // A collectible Godot game assembly may be reloaded in an Editor.
            // The gate/lease record uses only shared-framework types, not a
            // game-ALC type, and is installed only after complete verification.
            lock(string.Intern(StateKey)) VerifyAndLoad();
        } catch(Exception error) {
            OutputDebugString("GamePolicy: integrity verification refused.");
            bool loaded=OperatingSystem.IsWindows() && (GetModuleHandle("gua.dll")!=0 || GetModuleHandle("gua_runtime.dll")!=0);
            Console.Error.WriteLine("Gua Godot game integrity refused; stage="+stage+"; error="+error.GetType().Name+"; nativeLoaded="+loaded.ToString().ToLowerInvariant()+".");
            bool? editor=null;
            var game=typeof(GuaGameNativePolicy).Assembly;
            // No Godot native API here: exported game module initialization can
            // precede installation of its managed->native callback table.
            // In Godot 4.7 the trusted engine loader sets isCollectible from
            // its editorHint before loading the game (GodotPlugins.Main).
            var context=AssemblyLoadContext.GetLoadContext(game);
            var loader=context?.GetType();
            if(loader?.FullName=="GodotPlugins.PluginLoadContext" && loader.Assembly.GetName().Name=="GodotPlugins")
                editor=context!.IsCollectible;
            else if(!string.IsNullOrEmpty(game.Location) &&
                (context==AssemblyLoadContext.Default ||
                 (loader?.FullName=="Internal.Runtime.InteropServices.IsolatedComponentLoadContext" &&
                  loader.Assembly==typeof(object).Assembly && context?.IsCollectible==false)))
                editor=false;
            // Never terminate an Editor. A failed module initializer prevents
            // game assembly callbacks; restoration/restart is required.
            if(editor!=false) throw new InvalidOperationException("Gua game verification refused. Restore the selected SDK and restart the Godot Editor.");
            // Environment.Exit during a reverse Godot->CLR callback can enter
            // CLR teardown while native Godot is still on the callback stack.
            // Terminate only a confirmed game process, never an Editor.
            TerminateProcess(GetCurrentProcess(),loaded?79u:78u);
            throw new InvalidOperationException("Game termination failed.");
        }
    }

    internal static void RequireVerified()
    {
        if(!verified) throw new InvalidOperationException("Game-owned Gua verification has not succeeded.");
    }

    private static void VerifyAndLoad()
    {
        if(!OperatingSystem.IsWindows() || RuntimeInformation.ProcessArchitecture!=Architecture.X64)
            throw new PlatformNotSupportedException();
        stage="game-assembly-location";
        // Godot4.7 loads from streams, so Assembly.Location is empty. Its own
        // trusted loader (outside the Gua SDK set) exposes the actual file path.
        // Never ask an SDK assembly or environment override where to load from.
        var assembly=typeof(GuaGameNativePolicy).Assembly;
        string location=assembly.Location;
        if(string.IsNullOrEmpty(location)) {
            var context=AssemblyLoadContext.GetLoadContext(assembly);
            var loader=context?.GetType();
            if(loader?.FullName!="GodotPlugins.PluginLoadContext" || loader.Assembly.GetName().Name!="GodotPlugins")
                throw new PlatformNotSupportedException("Unsupported Godot managed loader.");
            location=loader.GetProperty("AssemblyLoadedPath")?.GetValue(context) as string??"";
        }
        if(string.IsNullOrEmpty(location)) throw new InvalidOperationException("A separate game assembly is required.");
        string directory=Path.GetFullPath(Path.GetDirectoryName(location)!);
        stage="existing-policy";
        string stamp=directory+"\n"+RemoteMode+"\n"+ProviderSource+"\n"+ProviderBuild+"\n"+
            string.Join("\n",Files.Select(f=>f.Name+":"+f.Sha256));
        if(AppContext.GetData(StateKey) is Tuple<string,FileStream[],nint[]> prior) {
            if(prior.Item1!=stamp || prior.Item2.Length!=Files.Length || prior.Item3.Length!=2 ||
               prior.Item2.Any(f=>!f.CanRead || f.SafeFileHandle.IsClosed) ||
               GetModuleHandle("gua.dll")!=prior.Item3[0] || GetModuleHandle("gua_runtime.dll")!=prior.Item3[1])
                throw new InvalidOperationException("The Editor's verified SDK differs from this game build. Restart required.");
            verified=true;
            return;
        }
        if(AppContext.GetData(StateKey)!=null || GetModuleHandle("gua.dll")!=0 || GetModuleHandle("gua_runtime.dll")!=0)
            throw new InvalidOperationException("Gua was loaded before the game policy.");
        for(string? path=directory;path!=null;path=Path.GetDirectoryName(path))
            if((File.GetAttributes(path)&FileAttributes.ReparsePoint)!=0) throw new IOException();
        var leases=new List<FileStream>();
        var modules=new List<nint>();
        foreach(var file in Files) {
            stage="check-"+file.Name;
            string path=Path.Combine(directory,file.Name);
            if((File.GetAttributes(path)&FileAttributes.ReparsePoint)!=0) throw new IOException();
            var stream=new FileStream(path,FileMode.Open,FileAccess.Read,FileShare.Read);
            leases.Add(stream);
            if(Convert.ToHexStringLower(SHA256.HashData(stream))!=file.Sha256) throw new InvalidDataException();
        }
        OutputDebugString("GamePolicy: verification succeeded before SDK initialization.");
        // All SDK managed/native files are locked and checked before either
        // native member is loaded. No Gua API/self-report participates.
        foreach(var file in Files) {
            if(file.Name is not ("gua.dll" or "gua_runtime.dll"))continue;
            stage="load-"+file.Name;
            nint module=LoadLibraryEx(Path.Combine(directory,file.Name),0,0x800);
            if(module==0)throw new System.ComponentModel.Win32Exception(Marshal.GetLastWin32Error());
            modules.Add(module);
        }
        AppContext.SetData("Gua.GameVerifiedCore",modules[0]);
        AppContext.SetData("Gua.GameVerifiedRuntime",modules[1]);
        AppContext.SetData(StateKey,Tuple.Create(stamp,leases.ToArray(),modules.ToArray()));
        verified=true;
    }

    [DllImport("kernel32.dll",CharSet=CharSet.Unicode,SetLastError=true)]
    private static extern nint LoadLibraryEx(string path,nint file,uint flags);
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode)]
    private static extern nint GetModuleHandle(string name);
    [DllImport("kernel32.dll")]
    private static extern nint GetCurrentProcess();
    [DllImport("kernel32.dll")]
    private static extern bool TerminateProcess(nint process,uint exitCode);
    [DllImport("kernel32.dll",CharSet=CharSet.Unicode)]
    private static extern void OutputDebugString(string text);
}
