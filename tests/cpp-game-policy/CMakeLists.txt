cmake_minimum_required(VERSION 3.20)
project(GuaProtectedConsumer LANGUAGES CXX)
find_package(GuaGameBuild CONFIG REQUIRED)
add_library(unrelated SHARED unrelated.cpp)
add_library(game_plugin SHARED plugin.cpp)
add_executable(game main.cpp)
target_link_libraries(game PRIVATE unrelated game_plugin)
set(hooks "")
if(LEGACY_HOOK)
  target_sources(game PRIVATE hooks.cpp)
  target_sources(game_plugin PRIVATE hooks.cpp)
  target_compile_definitions(game PRIVATE LEGACY_HOOK=1)
  target_link_options(game PRIVATE /DELAYLOAD:unrelated.dll)
  set(hooks NOTIFY_FALLBACK test_notify FAILURE_FALLBACK test_failure)
endif()
gua_protect_cpp_game(game MODE ${GAME_MODE} PLUGINS game_plugin ${hooks})
target_compile_definitions(game PRIVATE EXPECT_MODE="${GAME_MODE}")
# Deliberate violations: external SDK-dependent DLL and direct eager import.
add_library(legacy SHARED plugin.cpp)
target_include_directories(legacy PRIVATE "${GuaGameBuild_DIR}/include")
target_link_libraries(legacy PRIVATE "${GuaGameBuild_DIR}/off/gua.lib")
if(EARLY_PLUGIN)
  target_compile_definitions(game_plugin PRIVATE EARLY_PLUGIN=1)
endif()
if(EAGER_GAME)
  get_target_property(options game LINK_OPTIONS)
  list(REMOVE_ITEM options /DELAYLOAD:gua.dll)
  set_target_properties(game PROPERTIES LINK_OPTIONS "${options}")
endif()
if(EAGER_PLUGIN)
  get_target_property(options game_plugin LINK_OPTIONS)
  list(REMOVE_ITEM options /DELAYLOAD:gua.dll)
  set_target_properties(game_plugin PROPERTIES LINK_OPTIONS "${options}")
endif()
